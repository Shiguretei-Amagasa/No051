<script>

function generate() {

  const text = document.getElementById("text").value;

  const url = location.origin + "/ar.html?text=" + encodeURIComponent(text);


  document.getElementById("qrcode").innerHTML = "";


  new QRCode(document.getElementById("qrcode"), {

    text: url,

    width: 200,

    height: 200

  });

}
